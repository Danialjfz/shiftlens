\hd{Stack}
\begin{tabular}{@{}p{2.6cm}p{15.1cm}@{}}

Web server & FastAPI + uvicorn (only third-party deps); port 8200, probes 8200--8209\\

\hd{Data schemas}
\begin{tabular}{@{}p{3.4cm}p{14.3cm}@{}}
\texttt{workers.json} & \texttt{\{"workers":[\{worker\_id, name, role, shift,
base\_skill\}]\}} --- 8 workers; role $\in$ \{assembler, packer, inspector,
forklift-operator\}; \texttt{base\_skill} $\in [0.85, 1.20]$\\
\texttt{data/daily/}\newline\texttt{YYYY-MM-DD.json} & \texttt{\{date, environment\{noise\_db,
temperature\_c, air\_quality\_pm25\}, workers[\{worker\_id, sensors\{hr\_avg, hr\_max,
steps, station\_time\_min\}, detections\{ppe\_ok\_pct, phone\_events, idle\_events,
safety\_zone\_violations\}, survey\{fatigue, mood, perceived\_productivity, blockers\},
output\{units, defects, tasks\_completed\}\}]\}} --- 30 days from 2026-09-01; simulator
\texttt{random.Random(42)}\\
daily report & \texttt{out/daily\_reports/DATE/WID.json} + \texttt{.md}:
\texttt{\{worker\_id, date, metrics\{fatigue\_index, focus\_score, safety\_score, units,
defects\}, flags[], summary\}}\\
\texttt{out/monthly.json} & \texttt{\{period, team\_daily[]\{date, units, defects,
fatigue\_avg, noise\_db\}, workers[]\{worker\_id, name, units\_total, defect\_rate,
fatigue\_avg, focus\_avg, safety\_avg\}, correlations[]\{x, y, r, n, insight\},
findings[]\}}\\
\end{tabular}

\hd{Metrics (pure functions; $\operatorname{clamp}(x,0,1)=\min(\max(x,0),1)$) and flags (deterministic rules)}
\texttt{fatigue\_index} $= 0.6\cdot(\texttt{fatigue}/5) +
0.4\cdot\min(\texttt{hr\_avg}/100,\,1.2)/1.2$ \hfill
\texttt{focus\_score} $= \operatorname{clamp}(1 - 0.08\,\texttt{phone\_events} -
0.03\,\texttt{idle\_events},\,0,1)$\\[1pt]
\texttt{safety\_score} $= \operatorname{clamp}(\texttt{ppe\_ok\_pct}/100 -
0.25\,\texttt{safety\_zone\_violations},\,0,1)$ \hfill
\texttt{productivity} $=$ \texttt{units}, normalized per worker vs.\ own 30-day mean\\[1pt]
\texttt{high\_fatigue}: $\texttt{fatigue\_index} \ge 0.6$ \hfill
\texttt{low\_focus}: $\texttt{focus\_score} < 0.7$ \hfill
\texttt{safety\_risk}: $\texttt{safety\_score} < 0.85$ \hfill
\texttt{defect\_spike}: $\texttt{defects} \ge 2\times$ worker mean

\hd{Monthly analytics}
Pearson $r$ over all (worker, day) pairs, $n = 8 \times 30 = 240$; 7 pairs:
fatigue$\leftrightarrow$units, fatigue$\leftrightarrow$defects,
noise$\leftrightarrow$defects, temperature$\leftrightarrow$\texttt{ppe\_ok\_pct},
\texttt{phone\_events}$\leftrightarrow$units, mood$\leftrightarrow$units,
\texttt{focus\_score}$\leftrightarrow$units. Findings $=$ threshold splits (e.g.\ mean
defects on $>85$~dB vs.\ $\le 85$~dB days, \% delta) --- computed from data, never
hardcoded.

\hd{Agentic layer (v2) --- \texttt{shiftlens/agents/} (roles, tools, orchestrator, verifier)}
\begingroup\emergencystretch=2em\relax
Loop per task: plan $\to$ investigate $\to$ draft $\to$ verify $\to$ (revise once) $\to$
publish; LLM plans/drafts only --- heuristic plans + template drafts offline. Roles:
\textbf{orchestrator} (loop + trace), \textbf{planner} (ordered tool-call plan),
\textbf{investigator} (executes tools, collects evidence), \textbf{drafter} (evidence-only
prose), \textbf{verifier} (extracts every numeric claim, recomputes vs.\ tool data; fail
$\to$ one revision $\to$ template fallback). Tool registry (sole data channel, read-only):
\texttt{get\_worker\_day}, \texttt{get\_worker\_history}, \texttt{compare\_to\_team},
\texttt{get\_environment\_context}, \texttt{get\_flag\_rules}, \texttt{get\_correlations},
\texttt{get\_findings}, \texttt{get\_flagged\_days}. Artifacts: \texttt{investigation}
block in flagged reports; \texttt{narrative} + \texttt{agentic} stats block in
\texttt{monthly.json}; traces \texttt{out/traces/<date>\_<WID>.json} +
\texttt{out/traces/monthly.json}.\par\endgroup

\hd{API (FastAPI, port 8200; probes 8200--8209)}
\begin{tabular}{@{}p{3.6cm}p{14.6cm}@{}}

\texttt{GET /api/...} & \texttt{/api/monthly} (out/monthly.json; 404 until generated)

\end{tabular}

\hd{CLI (\texttt{../.venv/bin/python -m shiftlens.cli} from \texttt{ShiftLens/})}
\texttt{simulate} (generate dataset) $\cdot$ \texttt{report [--date D]} (daily reports)
$\cdot$ \texttt{monthly} (analytics) $\cdot$ \texttt{agents [--date D]} (agentic pass;
needs v1 artifacts) $\cdot$ \texttt{serve} (dashboard) $\cdot$ \texttt{demo} (simulate
$\to$ report $\to$ monthly $\to$ agents for first 3 flagged days, print findings + trace
summaries; \texttt{--all-agents} $=$ all flagged days)

\hd{Tests}
\texttt{tests/test\_smoke.py}: end-to-end demo into a tmp dir; asserts \texttt{monthly.json}
has the 7 correlations with $|r|>0$ for the planted signals, daily reports exist for all
workers, dashboard module imports; v2: traces exist with plan+evidence+verdict,
\texttt{narrative} non-empty, verifier pass rate $\ge 0.5$.
